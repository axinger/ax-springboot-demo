常用构建命令

# 构建所有模块
mvn clean install

# 跳过测试构建
mvn clean install -DskipTests

# 运行 Spring Boot 应用
cd <module-directory>
mvn spring-boot:run

# 运行测试
mvn test

# 批量更新版本
mvn versions:set -DnewVersion=x.x.x

环境配置

# 生产环境 (默认)
my-prd

# 开发环境
mvn clean install -Pmy-dev
